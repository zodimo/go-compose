Let's build a production-grade, zero-dependency, type-safe Form state engine in Go inspired by Angular Reactive Forms and Flutter's Form Architecture.

### Key Architectural Concepts

- **Composite Pattern:** Every element (`Control[T]`, `Group`, `Array`) implements the `FormNode` interface. Operations like `Validate()`, `MarkAsTouched()`, and `SetDisabled()` traverse the tree recursively.
- **Status Propagation:** Node status (`Valid`, `Invalid`, `Pending`, `Disabled`) flows upwards. If any child is `Invalid`, the parent `Group` is `Invalid`. If a parent is `Disabled`, its children report as `Disabled`.
- **Value Serialization:** Fully typed path lookup and value export into map structures (`map[string]any`).
- **Rx-like Subscriptions:** Fine-grained event streams (`OnStatusChange`, `OnValueChange`, `OnTouched`) so the UI layer only re-renders what actually changed.

### Core Domain Models

```go
package form

import (
	"sync"
)

type Status string

const (
	StatusValid    Status = "VALID"
	StatusInvalid  Status = "INVALID"
	StatusDisabled Status = "DISABLED"
)

// FormNode is the universal interface for leaf controls, groups, and arrays.
type FormNode interface {
	// Status & Validation
	Status() Status
	Validate() bool
	Errors() map[string]string
	
	// Lifecycle States
	IsTouched() bool
	IsDirty() bool
	IsPristine() bool
	IsEnabled() bool
	
	// Actions
	MarkAsTouched()
	MarkAsUntouched()
	MarkAsPristine()
	SetDisabled(disabled bool)
	Reset()
	
	// Export
	RawValue() any
	
	// Internal Tree Hooks
	setParent(parent FormNode)
	updateAncestors()
}
```

### Leaf Node: `Control[T]`

```go
type ValidatorFunc[T any] func(value T, node *Control[T]) string

type Control[T any] struct {
	mu            sync.RWMutex
	parent        FormNode
	initialValue  T
	value         T
	validators    []ValidatorFunc[T]
	errors        []string
	
	// States
	touched  bool
	dirty    bool
	disabled bool
	
	// Event Listeners
	valueListeners  []func(T)
	statusListeners []func(Status)
	touchListeners  []func(bool)
}

func NewControl[T any](initial T, validators ...ValidatorFunc[T]) *Control[T] {
	return &Control[T]{
		initialValue: initial,
		value:        initial,
		validators:   validators,
		errors:       make([]string, 0),
	}
}

// --- Value Management ---

func (c *Control[T]) Get() T {
	c.mu.RLock()
	defer c.mu.RUnlock()
	return c.value
}

func (c *Control[T]) Set(val T) {
	c.mu.Lock()
	if c.disabled {
		c.mu.Unlock()
		return
	}
	c.value = val
	c.dirty = true
	c.mu.Unlock()

	c.Validate()
	c.notifyValue(val)
	c.updateAncestors()
}

// --- Lifecycle & States ---

func (c *Control[T]) Status() Status {
	c.mu.RLock()
	defer c.mu.RUnlock()
	if c.disabled {
		return StatusDisabled
	}
	if len(c.errors) > 0 {
		return StatusInvalid
	}
	return StatusValid
}

func (c *Control[T]) MarkAsTouched() {
	c.mu.Lock()
	if c.touched {
		c.mu.Unlock()
		return
	}
	c.touched = true
	c.mu.Unlock()
	
	c.notifyTouch(true)
	c.updateAncestors()
}

func (c *Control[T]) MarkAsUntouched() {
	c.mu.Lock()
	c.touched = false
	c.mu.Unlock()
	c.notifyTouch(false)
}

func (c *Control[T]) MarkAsPristine() {
	c.mu.Lock()
	c.dirty = false
	c.mu.Unlock()
}

func (c *Control[T]) SetDisabled(disabled bool) {
	c.mu.Lock()
	c.disabled = disabled
	c.mu.Unlock()

	if !disabled {
		c.Validate()
	}
	c.notifyStatus(c.Status())
	c.updateAncestors()
}

func (c *Control[T]) IsTouched() bool  { c.mu.RLock(); defer c.mu.RUnlock(); return c.touched }
func (c *Control[T]) IsDirty() bool    { c.mu.RLock(); defer c.mu.RUnlock(); return c.dirty }
func (c *Control[T]) IsPristine() bool { c.mu.RLock(); defer c.mu.RUnlock(); return !c.dirty }
func (c *Control[T]) IsEnabled() bool  { c.mu.RLock(); defer c.mu.RUnlock(); return !c.disabled }

func (c *Control[T]) Validate() bool {
	c.mu.Lock()
	defer c.mu.Unlock()

	if c.disabled {
		c.errors = nil
		return true
	}

	c.errors = make([]string, 0)
	for _, v := range c.validators {
		if err := v(c.value, c); err != nil && err != "" {
			c.errors = append(c.errors, err)
		}
	}
	return len(c.errors) == 0
}

func (c *Control[T]) Errors() map[string]string {
	c.mu.RLock()
	defer c.mu.RUnlock()
	if len(c.errors) == 0 || c.disabled {
		return nil
	}
	return map[string]string{"error": c.errors[0]}
}

func (c *Control[T]) Reset() {
	c.mu.Lock()
	c.value = c.initialValue
	c.dirty = false
	c.touched = false
	c.errors = nil
	c.mu.Unlock()

	c.notifyValue(c.initialValue)
	c.notifyTouch(false)
	c.updateAncestors()
}

func (c *Control[T]) RawValue() any { return c.Get() }

// --- Event Subscription ---

func (c *Control[T]) OnValueChange(fn func(T))   { c.valueListeners = append(c.valueListeners, fn) }
func (c *Control[T]) OnStatusChange(fn func(Status)) { c.statusListeners = append(c.statusListeners, fn) }
func (c *Control[T]) OnTouchChange(fn func(bool))  { c.touchListeners = append(c.touchListeners, fn) }

func (c *Control[T]) notifyValue(v T)        { for _, fn := range c.valueListeners { fn(v) } }
func (c *Control[T]) notifyStatus(s Status)  { for _, fn := range c.statusListeners { fn(s) } }
func (c *Control[T]) notifyTouch(t bool)     { for _, fn := range c.touchListeners { fn(t) } }

func (c *Control[T]) setParent(p FormNode) { c.parent = p }
func (c *Control[T]) updateAncestors() {
	if c.parent != nil {
		c.parent.updateAncestors()
	}
}
```

### Composite Branch Node: `Group`

```go
type Group struct {
	mu       sync.RWMutex
	parent   FormNode
	children map[string]FormNode
	disabled bool
}

func NewGroup(children map[string]FormNode) *Group {
	g := &Group{
		children: children,
	}
	for _, child := range children {
		child.setParent(g)
	}
	return g
}

func (g *Group) Get(path string) FormNode {
	g.mu.RLock()
	defer g.mu.RUnlock()
	return g.children[path]
}

func (g *Group) Status() Status {
	g.mu.RLock()
	defer g.mu.RUnlock()

	if g.disabled {
		return StatusDisabled
	}

	allDisabled := true
	for _, child := range g.children {
		status := child.Status()
		if status == StatusInvalid {
			return StatusInvalid
		}
		if status != StatusDisabled {
			allDisabled = false
		}
	}

	if allDisabled {
		return StatusDisabled
	}
	return StatusValid
}

func (g *Group) Validate() bool {
	g.mu.RLock()
	defer g.mu.RUnlock()

	if g.disabled {
		return true
	}

	valid := true
	for _, child := range g.children {
		if !child.Validate() {
			valid = false
		}
	}
	return valid
}

func (g *Group) IsTouched() bool {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		if child.IsTouched() {
			return true
		}
	}
	return false
}

func (g *Group) IsDirty() bool {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		if child.IsDirty() {
			return true
		}
	}
	return false
}

func (g *Group) IsPristine() bool { return !g.IsDirty() }
func (g *Group) IsEnabled() bool  { return !g.disabled }

func (g *Group) MarkAsTouched() {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		child.MarkAsTouched()
	}
}

func (g *Group) MarkAsUntouched() {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		child.MarkAsUntouched()
	}
}

func (g *Group) MarkAsPristine() {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		child.MarkAsPristine()
	}
}

func (g *Group) SetDisabled(disabled bool) {
	g.mu.Lock()
	g.disabled = disabled
	g.mu.Unlock()

	for _, child := range g.children {
		child.SetDisabled(disabled)
	}
	g.updateAncestors()
}

func (g *Group) Reset() {
	g.mu.RLock()
	defer g.mu.RUnlock()
	for _, child := range g.children {
		child.Reset()
	}
}

func (g *Group) Errors() map[string]string {
	g.mu.RLock()
	defer g.mu.RUnlock()

	if g.disabled {
		return nil
	}

	errs := make(map[string]string)
	for name, child := range g.children {
		for k, v := range child.Errors() {
			errs[name+"."+k] = v
		}
	}
	if len(errs) == 0 {
		return nil
	}
	return errs
}

func (g *Group) RawValue() any {
	g.mu.RLock()
	defer g.mu.RUnlock()

	valMap := make(map[string]any)
	for name, child := range g.children {
		if child.IsEnabled() {
			valMap[name] = child.RawValue()
		}
	}
	return valMap
}

func (g *Group) setParent(p FormNode) { g.parent = p }
func (g *Group) updateAncestors() {
	if g.parent != nil {
		g.parent.updateAncestors()
	}
}
```

### Built-in Production Validators

```go
func Required[T comparable](zero T) ValidatorFunc[T] {
	return func(value T, node *Control[T]) string {
		if value == zero {
			return "field is required"
		}
		return ""
	}
}

func MinLength(min int) ValidatorFunc[string] {
	return func(value string, node *Control[string]) string {
		if len(value) < min {
			return "minimum length not met"
		}
		return ""
	}
}
```

### Production Setup & Execution

```go
package main

import (
	"fmt"
)

func main() {
	// 1. Instantiate Controls with Validation
	nameCtrl := NewControl("", Required(""))
	isShippingSameCtrl := NewControl(true)
	
	streetCtrl := NewControl("", Required(""))
	cityCtrl := NewControl("", Required(""))

	// 2. Build Nested Form Graph
	addressGroup := NewGroup(map[string]FormNode{
		"street": streetCtrl,
		"city":   cityCtrl,
	})

	form := NewGroup(map[string]FormNode{
		"name":           nameCtrl,
		"sameAsShipping": isShippingSameCtrl,
		"billingAddress": addressGroup,
	})

	// 3. Dynamic Field Interdependency (Disable billing if sameAsShipping is true)
	isShippingSameCtrl.OnValueChange(func(same bool) {
		addressGroup.SetDisabled(same)
	})

	// Initial State Check
	isShippingSameCtrl.Set(true) // Triggers billingAddress -> Disabled
	
	fmt.Println("--- Initial Form State ---")
	fmt.Println("Form Status (Billing disabled):", form.Status()) // VALID because billing is disabled!
	fmt.Println("Is Form Touched?:", form.IsTouched())            // false
	fmt.Println("Is Form Dirty?:", form.IsDirty())                // false

	// User Toggles Shipping off
	fmt.Println("\n--- User Unticks 'Same as Shipping' ---")
	isShippingSameCtrl.Set(false) // Enables billingAddress group
	
	fmt.Println("Form Status (Billing enabled & empty):", form.Status()) // INVALID
	fmt.Println("Errors:", form.Errors())

	// Simulate Blur/Focus Interaction
	fmt.Println("\n--- User Touches Street Field ---")
	streetCtrl.MarkAsTouched()
	fmt.Println("Is Street Touched?:", streetCtrl.IsTouched())
	fmt.Println("Is Form Touched?:", form.IsTouched()) // True via bubble-up

	// User Completes the Form
	fmt.Println("\n--- Filling out Form ---")
	nameCtrl.Set("Jane Doe")
	streetCtrl.Set("42 Wallaby Way")
	cityCtrl.Set("Sydney")

	fmt.Println("Form Status:", form.Status())       // VALID
	fmt.Println("Form Raw Export:", form.RawValue()) // Serialized map
}
```
